import React, { useState, useEffect } from "react";
import {
  BrowserRouter,
  Routes,
  Route,
  Link,
  Navigate,
  useParams
} from "react-router-dom";

function Home() {
  return <h2>Home Page</h2>;
}

// EXPERIMENT 4: E-COMMERCE APPLICATION

function Products() {
  const products = [
    { id: 101, name: "Laptop", price: 50000 },
    { id: 102, name: "Mobile", price: 20000 },
    { id: 103, name: "Shoes", price: 1500 }
  ];

  return (
    <div>
      <h2>Products Page</h2>
      {products.map((p) => (
        <div key={p.id}>
          <p>{p.name} - ₹{p.price}</p>
          <Link to={"/products/" + p.id}>View Details</Link>
          <hr />
        </div>
      ))}
    </div>
  );
}

function ProductDetails() {
  const { id } = useParams();

  return (
    <div>
      <h2>Product Details</h2>
      <p>Product ID: {id}</p>
    </div>
  );
}

function Cart() {
  return <h2>Cart Page</h2>;
}

function Contact() {
  return <h2>Contact Page</h2>;
}

function Login({ setLoggedIn }) {
  return (
    <div>
      <h2>Login Page</h2>
      <button onClick={() => setLoggedIn(true)}>
        Login
      </button>
    </div>
  );
}

function ProtectedRoute({ loggedIn, children }) {
  return loggedIn ? children : <Navigate to="/login" />;
}

function NotFound() {
  return <h2>404 Page Not Found</h2>;
}

// EXPERIMENT 5: DYNAMIC REACT FORM

function DynamicForm() {
  const fields = [
    { name: "name", label: "Name", type: "text" },
    { name: "email", label: "Email", type: "email" },
    { name: "mobile", label: "Mobile Number", type: "text" },
    { name: "age", label: "Age", type: "number" },
    { name: "city", label: "City", type: "text" }
  ];

  const [formData, setFormData] = useState({});
  const [submitted, setSubmitted] = useState(null);

  function handleChange(e) {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value
    });
  }

  function handleSubmit(e) {
    e.preventDefault();
    setSubmitted({ ...formData });
  }

  return (
    <div>
      <h2>Dynamic Form</h2>

      <form onSubmit={handleSubmit}>
        {fields.map((field) => (
          <div key={field.name}>
            <label>{field.label}: </label>
            <input
              type={field.type}
              name={field.name}
              value={formData[field.name] || ""}
              onChange={handleChange}
              required
            />
            <br /><br />
          </div>
        ))}

        <button type="submit">Submit</button>
      </form>

      {submitted && (
        <div>
          <h3>Submitted Details</h3>
          {fields.map((field) => (
            <p key={field.name}>
              {field.label}: {submitted[field.name]}
            </p>
          ))}
        </div>
      )}
    </div>
  );
}

// EXPERIMENT 6: SPA WITH API FETCHING

function Users() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch("https://jsonplaceholder.typicode.com/users")
      .then((response) => response.json())
      .then((data) => {
        setUsers(data);
        setLoading(false);
      })
      .catch(() => {
        setLoading(false);
      });
  }, []);

  if (loading) {
    return <h3>Loading...</h3>;
  }

  return (
    <div>
      <h2>Users List</h2>
      {users.length === 0 ? (
        <p>Unable to load users. Check your internet connection.</p>
      ) : (
        <ul>
          {users.map((user) => (
            <li key={user.id}>{user.name}</li>
          ))}
        </ul>
      )}
    </div>
  );
}

// MAIN APPLICATION

function App() {
  const [loggedIn, setLoggedIn] = useState(false);

  return (
    <BrowserRouter>
      <nav>
        <Link to="/">Home</Link>{" | "}
        <Link to="/products">Products</Link>{" | "}
        <Link to="/cart">Cart</Link>{" | "}
        <Link to="/contact">Contact</Link>{" | "}
        <Link to="/login">Login</Link>{" | "}
        <Link to="/form">Experiment 5 - Form</Link>{" | "}
        <Link to="/users">Experiment 6 - Users</Link>
      </nav>

      <hr />

      <Routes>
        <Route path="/" element={<Home />} />

        <Route path="/products" element={<Products />} />
        <Route
          path="/products/:id"
          element={<ProductDetails />}
        />

        <Route
          path="/cart"
          element={
            <ProtectedRoute loggedIn={loggedIn}>
              <Cart />
            </ProtectedRoute>
          }
        />

        <Route path="/contact" element={<Contact />} />
        <Route
          path="/login"
          element={<Login setLoggedIn={setLoggedIn} />}
        />

        <Route path="/form" element={<DynamicForm />} />
        <Route path="/users" element={<Users />} />

        <Route path="*" element={<NotFound />} />
      </Routes>
    </BrowserRouter>
  );
}

export default App;
